       @component
       class HelloHttp(HttpPlugin):
           @get('/hello/(?P<name>.+)')
           def get_page(self, http_context, name=None):
               context.add_header('Content-Type', 'text/plain')
               context.respond_ok()
               return 'Hello, f"{name}"!'
